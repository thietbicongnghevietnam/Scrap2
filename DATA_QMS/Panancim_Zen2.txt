--dieu kien: >= luot kitting cuoi cung (top2) and < time san xuat 
--cach lay top 2 dau tien  : 2026-05-19 02:02:08.493 => sua lai thanh TOP 1
WITH cte AS
(
    SELECT
        ktld.KittingListID,
        MAX(ktld.UpdatedDate) AS MaxUpdated
    FROM tblKittingListDetail (nolock) ktld
    LEFT JOIN tblKittingList ktl
        ON ktld.KittingListID = ktl.ID
    WHERE ktl.Model = 'SA-AU1F-S-T-V4'
	and ktl.KittingType <> 3 --khac kitting outside
      --AND ktl.Deliverydate = '2026-08-20'
      AND ktld.UpdatedDate < '2026-08-22 6:10:43'
      AND ktld.SupplyStatus = 1
      AND ktld.Line = 'IS1'
    GROUP BY ktld.KittingListID
),
rn AS
(
    SELECT *,
           ROW_NUMBER() OVER (
               ORDER BY MaxUpdated DESC
           ) AS RowNum
    FROM cte
)
SELECT
    KittingListID,
    MaxUpdated AS UpdatedDate
FROM rn
WHERE RowNum = 1
